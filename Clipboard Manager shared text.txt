import logging
import time
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any

from app.agent.planner import ExecutionPlan, PlanStep, PlanStepType
from app.agent.tools.memory_tool import memory_tool_handler
from app.agent.tools.reminder_tool import reminder_tool_handler
from app.agent.tools.registry import ToolResult, tool_registry


logger = logging.getLogger("sara.agent.executor")


# ============================================================
# EXECUTION RESULT
# ============================================================

@dataclass
class ExecutionResult:
    success: bool
    response_text: str = ""
    steps: list[dict[str, Any]] = field(default_factory=list)
    error: str | None = None
    metadata: dict[str, Any] = field(default_factory=dict)

    @property
    def completed_steps(self) -> int:
        return sum(
            1
            for step in self.steps
            if step.get("completed") is True
        )

    @property
    def failed_steps(self) -> int:
        return sum(
            1
            for step in self.steps
            if step.get("error")
        )

    def to_dict(self) -> dict[str, Any]:
        return {
            "success": self.success,
            "response_text": self.response_text,
            "steps": self.steps,
            "error": self.error,
            "metadata": self.metadata,
            "completed_steps": self.completed_steps,
            "failed_steps": self.failed_steps,
        }


# ============================================================
# SARA EXECUTOR
# ============================================================

class SaraExecutor:
    """
    Planner yaratgan ExecutionPlan'ni amalda bajaradi.

    Executor vazifalari:

    - memory retrieval
    - group memory retrieval
    - relationship/session context
    - memory save
    - reminder create
    - tool execution
    - response generation resultini qabul qilish
    - Telegram'ga javob yuborish
    - plan lifecycle boshqarish
    """

    def __init__(self) -> None:
        self._executions = 0
        self._successful_executions = 0
        self._failed_executions = 0

    # ========================================================
    # MAIN EXECUTION
    # ========================================================

    async def execute(
        self,
        plan: ExecutionPlan,
    ) -> ExecutionResult:

        started = time.monotonic()

        if plan is None:
            return ExecutionResult(
                success=False,
                error="execution_plan_missing",
            )

        try:
            validation = self._validate_plan(plan)

            if not validation["valid"]:
                return ExecutionResult(
                    success=False,
                    error=validation["error"],
                )

            self._executions += 1

            response_text = ""
            step_results: list[dict[str, Any]] = []

            while True:
                step = plan.get_current_step()

                if step is None:
                    break

                try:
                    result = await self.execute_step(
                        step=step,
                        plan=plan,
                    )

                    if isinstance(result, ToolResult):
                        success = result.success
                        result_data = result.result
                        error = result.error
                    elif isinstance(result, dict):
                        success = bool(result.get("success", True))
                        result_data = result
                        error = result.get("error")
                    else:
                        success = True
                        result_data = result
                        error = None

                    if success:
                        plan.complete_current_step(result_data)

                        if isinstance(result_data, dict):
                            candidate = result_data.get("text")

                            if candidate:
                                response_text = str(candidate).strip()

                            candidate = result_data.get("response_text")

                            if candidate:
                                response_text = str(candidate).strip()

                        step_results.append(
                            {
                                "step_id": step.id,
                                "step_type": step.step_type.value,
                                "completed": True,
                                "result": self._safe_result(result_data),
                            }
                        )

                    else:
                        plan.fail_current_step(
                            str(error or "step_failed")
                        )

                        step_results.append(
                            {
                                "step_id": step.id,
                                "step_type": step.step_type.value,
                                "completed": False,
                                "error": str(
                                    error or "step_failed"
                                ),
                            }
                        )

                        if step.required:
                            plan.cancel(
                                f"required_step_failed:{step.id}"
                            )

                            self._failed_executions += 1

                            return ExecutionResult(
                                success=False,
                                response_text=response_text,
                                steps=step_results,
                                error=str(
                                    error or "required_step_failed"
                                ),
                                metadata={
                                    "plan_id": plan.plan_id,
                                    "duration_seconds": round(
                                        time.monotonic() - started,
                                        4,
                                    ),
                                },
                            )

                except Exception as exc:
                    logger.exception(
                        "Executor step failed | plan=%s | step=%s",
                        plan.plan_id,
                        step.id,
                    )

                    plan.fail_current_step(str(exc))

                    step_results.append(
                        {
                            "step_id": step.id,
                            "step_type": step.step_type.value,
                            "completed": False,
                            "error": str(exc),
                        }
                    )

                    if step.required:
                        plan.cancel(
                            f"exception_in_required_step:{step.id}"
                        )

                        self._failed_executions += 1

                        return ExecutionResult(
                            success=False,
                            response_text=response_text,
                            steps=step_results,
                            error=str(exc),
                            metadata={
                                "plan_id": plan.plan_id,
                                "duration_seconds": round(
                                    time.monotonic() - started,
                                    4,
                                ),
                            },
                        )

            # ------------------------------------------------
            # FINAL RESPONSE
            # ------------------------------------------------

            if not response_text:
                response_text = str(
                    plan.metadata.get(
                        "response_text",
                        "",
                    )
                    or ""
                ).strip()

            success = not plan.cancelled

            if success:
                self._successful_executions += 1
            else:
                self._failed_executions += 1

            return ExecutionResult(
                success=success,
                response_text=response_text,
                steps=step_results,
                error=plan.cancel_reason,
                metadata={
                    "plan_id": plan.plan_id,
                    "action": plan.action,
                    "progress": plan.progress(),
                    "duration_seconds": round(
                        time.monotonic() - started,
                        4,
                    ),
                },
            )

        except Exception as exc:
            self._failed_executions += 1

            logger.exception(
                "Executor failed | plan=%s",
                getattr(plan, "plan_id", "unknown"),
            )

            return ExecutionResult(
                success=False,
                error=str(exc),
                metadata={
                    "duration_seconds": round(
                        time.monotonic() - started,
                        4,
                    )
                },
            )

    # ========================================================
    # EXECUTE SINGLE STEP
    # ========================================================

    async def execute_step(
        self,
        *,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> Any:

        step_type = step.step_type

        if step_type == PlanStepType.ANALYZE:
            return await self._handle_analyze(
                step,
                plan,
            )

        if step_type == PlanStepType.RETRIEVE_MEMORY:
            return await self._handle_retrieve_memory(
                step,
                plan,
            )

        if step_type == PlanStepType.RETRIEVE_GROUP_MEMORY:
            return await self._handle_retrieve_group_memory(
                step,
                plan,
            )

        if step_type == PlanStepType.CHECK_RELATIONSHIP:
            return await self._handle_check_relationship(
                step,
                plan,
            )

        if step_type == PlanStepType.CHECK_SESSION_STATE:
            return await self._handle_check_session_state(
                step,
                plan,
            )

        if step_type == PlanStepType.GENERATE_RESPONSE:
            return await self._handle_generate_response(
                step,
                plan,
            )

        if step_type == PlanStepType.ASK_CLARIFICATION:
            return await self._handle_ask_clarification(
                step,
                plan,
            )

        if step_type == PlanStepType.SAVE_MEMORY:
            return await self._handle_save_memory(
                step,
                plan,
            )

        if step_type == PlanStepType.CREATE_REMINDER:
            return await self._handle_create_reminder(
                step,
                plan,
            )

        if step_type == PlanStepType.USE_TOOL:
            return await self._handle_use_tool(
                step,
                plan,
            )

        if step_type == PlanStepType.SEND_RESPONSE:
            return await self._handle_send_response(
                step,
                plan,
            )

        if step_type == PlanStepType.FINISH:
            return await self._handle_finish(
                step,
                plan,
            )

        return {
            "success": False,
            "error": f"unsupported_step_type:{step_type}",
        }

    # ========================================================
    # ANALYZE
    # ========================================================

    async def _handle_analyze(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        return {
            "success": True,
            "action": plan.action,
            "user_text": plan.user_text,
            "plan_id": plan.plan_id,
        }

    # ========================================================
    # RETRIEVE USER MEMORY
    # ========================================================

    async def _handle_retrieve_memory(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        user_id = args.get(
            "user_telegram_id"
        )

        if user_id is None:
            return {
                "success": False,
                "error": "user_telegram_id_missing",
            }

        query = str(
            args.get(
                "query",
                plan.user_text,
            )
            or ""
        ).strip()

        limit = self._safe_int(
            args.get("limit", 10),
            10,
            1,
            50,
        )

        result = await memory_tool_handler(
            operation="search_user",
            user_telegram_id=int(user_id),
            query=query,
            limit=limit,
        )

        return self._normalize_tool_result(
            result,
            "memory_retrieval_failed",
        )

    # ========================================================
    # RETRIEVE GROUP MEMORY
    # ========================================================

    async def _handle_retrieve_group_memory(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        group_id = args.get(
            "group_telegram_id"
        )

        if group_id is None:
            return {
                "success": False,
                "error": "group_telegram_id_missing",
            }

        query = str(
            args.get(
                "query",
                plan.user_text,
            )
            or ""
        ).strip()

        limit = self._safe_int(
            args.get("limit", 10),
            10,
            1,
            50,
        )

        result = await memory_tool_handler(
            operation="search_group",
            group_telegram_id=int(group_id),
            query=query,
            limit=limit,
        )

        return self._normalize_tool_result(
            result,
            "group_memory_retrieval_failed",
        )

    # ========================================================
    # RELATIONSHIP
    # ========================================================

    async def _handle_check_relationship(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        return {
            "success": True,
            "user_a": args.get("user_a"),
            "user_b": args.get("user_b"),
            "relationship_context": plan.metadata.get(
                "relationship_context"
            ),
        }

    # ========================================================
    # SESSION STATE
    # ========================================================

    async def _handle_check_session_state(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        return {
            "success": True,
            "session_state": plan.metadata.get(
                "session_state",
                {},
            ),
            "emotional_state": plan.metadata.get(
                "emotional_state",
                {},
            ),
        }

    # ========================================================
    # GENERATE RESPONSE
    # ========================================================

    async def _handle_generate_response(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        response_text = str(
            args.get(
                "response_text",
                "",
            )
            or plan.metadata.get(
                "response_text",
                "",
            )
            or ""
        ).strip()

        # Orchestrator odatda AI javobini oldindan yaratadi.
        # Shuning uchun Executor bu yerda ikkinchi marta
        # OpenRouter chaqirmaydi.

        if not response_text:
            return {
                "success": False,
                "error": "response_text_missing",
            }

        return {
            "success": True,
            "response_text": response_text,
            "text": response_text,
        }

    # ========================================================
    # ASK CLARIFICATION
    # ========================================================

    async def _handle_ask_clarification(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        response_text = str(
            args.get(
                "response_text",
                "",
            )
            or plan.metadata.get(
                "response_text",
                "",
            )
            or ""
        ).strip()

        if not response_text:
            response_text = (
                "Aniqroq tushuntirib bera olasanmi?"
            )

        return {
            "success": True,
            "response_text": response_text,
            "text": response_text,
        }

    # ========================================================
    # SAVE MEMORY
    # ========================================================

    async def _handle_save_memory(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        content = str(
            args.get(
                "content",
                plan.user_text,
            )
            or ""
        ).strip()

        if not content:
            return {
                "success": False,
                "error": "memory_content_missing",
            }

        memory_type = str(
            args.get(
                "memory_type",
                "important_fact",
            )
            or "important_fact"
        ).strip()

        user_id = args.get(
            "user_telegram_id"
        )

        group_id = args.get(
            "group_telegram_id"
        )

        importance = self._safe_float(
            args.get("importance", 0.7),
            0.7,
            0.0,
            1.0,
        )

        confidence = self._safe_float(
            args.get("confidence", 0.8),
            0.8,
            0.0,
            1.0,
        )

        source_message_id = args.get(
            "source_message_id"
        )

        # -----------------------------------------------
        # GROUP MEMORY
        # -----------------------------------------------

        if group_id is not None:
            result = await memory_tool_handler(
                operation="save_group",
                group_telegram_id=int(group_id),
                memory_type=memory_type,
                content=content,
                importance=importance,
                confidence=confidence,
                source_message_id=source_message_id,
            )

            return self._normalize_tool_result(
                result,
                "group_memory_save_failed",
            )

        # -----------------------------------------------
        # USER MEMORY
        # -----------------------------------------------

        if user_id is not None:
            result = await memory_tool_handler(
                operation="save_user",
                user_telegram_id=int(user_id),
                memory_type=memory_type,
                content=content,
                importance=importance,
                confidence=confidence,
                source_message_id=source_message_id,
            )

            return self._normalize_tool_result(
                result,
                "memory_save_failed",
            )

        return {
            "success": False,
            "error": "memory_owner_missing",
        }

    # ========================================================
    # CREATE REMINDER
    # ========================================================

    async def _handle_create_reminder(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        owner_id = args.get(
            "owner_telegram_id"
        )

        chat_id = args.get(
            "chat_id"
        )

        text = str(
            args.get(
                "text",
                plan.user_text,
            )
            or ""
        ).strip()

        if owner_id is None:
            return {
                "success": False,
                "error": "owner_telegram_id_missing",
            }

        if chat_id is None:
            return {
                "success": False,
                "error": "chat_id_missing",
            }

        if not text:
            return {
                "success": False,
                "error": "reminder_text_missing",
            }

        result = await reminder_tool_handler(
            operation="create",
            owner_telegram_id=int(owner_id),
            chat_id=int(chat_id),
            text=text,
        )

        return self._normalize_tool_result(
            result,
            "reminder_creation_failed",
        )

    # ========================================================
    # USE TOOL
    # ========================================================

    async def _handle_use_tool(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> Any:

        args = step.arguments or {}

        tool_name = str(
            args.get(
                "tool_name",
                "",
            )
            or ""
        ).strip()

        tool_arguments = args.get(
            "tool_arguments",
            {},
        )

        if not tool_name:
            return {
                "success": False,
                "error": "tool_name_missing",
            }

        if not isinstance(
            tool_arguments,
            dict,
        ):
            tool_arguments = {}

        try:
            result = await tool_registry.execute(
                tool_name,
                **tool_arguments,
            )
        except TypeError:
            # Ayrim ToolRegistry versiyalarida
            # arguments dict sifatida qabul qilinishi mumkin.
            result = await tool_registry.execute(
                tool_name,
                tool_arguments,
            )

        return self._normalize_tool_result(
            result,
            f"tool_failed:{tool_name}",
        )

    # ========================================================
    # SEND TELEGRAM RESPONSE
    # ========================================================

    async def _handle_send_response(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        args = step.arguments or {}

        chat_id = args.get(
            "chat_id",
            plan.metadata.get("chat_id"),
        )

        reply_to_message_id = args.get(
            "reply_to_message_id",
            plan.metadata.get(
                "reply_to_message_id"
            ),
        )

        response_text = str(
            args.get(
                "response_text",
                "",
            )
            or plan.metadata.get(
                "response_text",
                "",
            )
            or ""
        ).strip()

        if not response_text:
            return {
                "success": False,
                "error": "send_response_text_missing",
            }

        if chat_id is None:
            return {
                "success": False,
                "error": "send_chat_id_missing",
            }

        try:
            result = await tool_registry.execute(
                "telegram_send_message",
                chat_id=int(chat_id),
                text=response_text,
                reply_to_message_id=(
                    int(reply_to_message_id)
                    if reply_to_message_id is not None
                    else None
                ),
            )
        except TypeError:
            result = await tool_registry.execute(
                "telegram_send_message",
                {
                    "chat_id": int(chat_id),
                    "text": response_text,
                    "reply_to_message_id": (
                        int(reply_to_message_id)
                        if reply_to_message_id is not None
                        else None
                    ),
                },
            )

        normalized = self._normalize_tool_result(
            result,
            "telegram_send_failed",
        )

        if normalized.get("success"):
            normalized["response_text"] = response_text
            normalized["text"] = response_text

        return normalized

    # ========================================================
    # FINISH
    # ========================================================

    async def _handle_finish(
        self,
        step: PlanStep,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        return {
            "success": True,
            "finished": True,
            "plan_id": plan.plan_id,
        }

    # ========================================================
    # VALIDATION
    # ========================================================

    def _validate_plan(
        self,
        plan: ExecutionPlan,
    ) -> dict[str, Any]:

        if not plan.plan_id:
            return {
                "valid": False,
                "error": "plan_id_missing",
            }

        if not plan.action:
            return {
                "valid": False,
                "error": "plan_action_missing",
            }

        if not isinstance(
            plan.steps,
            list,
        ):
            return {
                "valid": False,
                "error": "plan_steps_invalid",
            }

        if not plan.steps:
            return {
                "valid": False,
                "error": "plan_has_no_steps",
            }

        return {
            "valid": True,
            "error": None,
        }

    # ========================================================
    # HELPERS
    # ========================================================

    @staticmethod
    def _normalize_tool_result(
        result: Any,
        default_error: str,
    ) -> dict[str, Any]:

        if isinstance(
            result,
            ToolResult,
        ):
            return {
                "success": result.success,
                "result": result.result,
                "error": (
                    result.error
                    or (
                        None
                        if result.success
                        else default_error
                    )
                ),
                "tool_name": result.tool_name,
                "duration_seconds": (
                    result.duration_seconds
                ),
            }

        if isinstance(
            result,
            dict,
        ):
            success = bool(
                result.get(
                    "success",
                    True,
                )
            )

            return {
                **result,
                "success": success,
                "error": (
                    result.get("error")
                    if not success
                    else result.get("error")
                ),
            }

        return {
            "success": True,
            "result": result,
        }

    @staticmethod
    def _safe_result(
        value: Any,
    ) -> Any:

        if isinstance(
            value,
            (str, int, float, bool),
        ):
            return value

        if value is None:
            return None

        if isinstance(
            value,
            dict,
        ):
            output: dict[str, Any] = {}

            for key, item in value.items():
                if isinstance(
                    item,
                    (str, int, float, bool),
                ) or item is None:
                    output[str(key)] = item
                elif isinstance(
                    item,
                    (list, tuple),
                ):
                    output[str(key)] = [
                        str(x)
                        for x in item
                    ]
                else:
                    output[str(key)] = str(item)

            return output

        if isinstance(
            value,
            (list, tuple),
        ):
            return [
                str(item)
                for item in value
            ]

        return str(value)

    @staticmethod
    def _safe_int(
        value: Any,
        default: int,
        minimum: int,
        maximum: int,
    ) -> int:

        try:
            value = int(value)
        except (
            TypeError,
            ValueError,
        ):
            value = default

        return max(
            minimum,
            min(
                maximum,
                value,
            ),
        )

    @staticmethod
    def _safe_float(
        value: Any,
        default: float,
        minimum: float,
        maximum: float,
    ) -> float:

        try:
            value = float(value)
        except (
            TypeError,
            ValueError,
        ):
            value = default

        return max(
            minimum,
            min(
                maximum,
                value,
            ),
        )

    # ========================================================
    # STATS
    # ========================================================

    def stats(self) -> dict[str, Any]:

        success_rate = 0.0

        if self._executions:
            success_rate = (
                self._successful_executions
                / self._executions
            )

        return {
            "executions": self._executions,
            "successful_executions": (
                self._successful_executions
            ),
            "failed_executions": (
                self._failed_executions
            ),
            "success_rate": round(
                success_rate,
                4,
            ),
        }

    def reset_stats(self) -> None:
        self._executions = 0
        self._successful_executions = 0
        self._failed_executions = 0
