def get_command_args(message: Message) -> str:
    """
    Telegram commanddan keyingi argumentlarni oladi.

    Masalan:

        /memory search minecraft

    natija:

        search minecraft
    """

    text = message.text or ""

    parts = text.split(maxsplit=1)

    if len(parts) < 2:
        return ""

    return parts[1].strip()


def split_args(message: Message) -> list[str]:
    args = get_command_args(message)

    if not args:
        return []

    return args.split()


def format_memory(memory) -> str:
    """
    UserMemory obyektini foydalanuvchiga
    tushunarli formatga o'tkazadi.
    """

    return (
        f"🧠 <b>#{memory.id}</b>\n"
        f"📌 Type: <code>{memory.memory_type}</code>\n"
        f"⭐ Importance: <code>{memory.importance}/100</code>\n"
        f"🎯 Confidence: <code>{memory.confidence:.2f}</code>\n"
        f"📝 {memory.content}"
    )


def format_group_memory(memory) -> str:
    """
    GroupMemory obyektini formatlaydi.
    """

    return (
        f"🏷 <b>#{memory.id}</b>\n"
        f"📌 Type: <code>{memory.memory_type}</code>\n"
        f"⭐ Importance: <code>{memory.importance}/100</code>\n"
        f"🎯 Confidence: <code>{memory.confidence:.2f}</code>\n"
        f"📝 {memory.content}"
    )


async def is_group_admin(message: Message) -> bool:
    """
    User guruh adminimi yoki yo'qmi.
    """

    if message.chat.type not in {"group", "supergroup"}:
        return False

    if message.from_user is None:
        return False

    return await permission_service.can_manage_group_async(
        bot=message.bot,
        chat_id=message.chat.id,
        user_id=message.from_user.id,
    )

@router.message(Command("memory"))
async def command_memory(message: Message) -> None:
    """
    /memory
    /memory list
    /memory search ...
    /memory stats
    """

    if message.from_user is None:
        return

    args = get_command_args(message)

    # --------------------------------------------------------
    # /memory
    # /memory list
    # --------------------------------------------------------

    if not args or args.lower() == "list":

        memories = await memory_manager.get_user_memories(
            user_telegram_id=message.from_user.id,
            limit=settings.max_memory_results,
        )

        if not memories:
            await message.answer(
                "🧠 Hozircha sen haqingda saqlangan "
                "xotira yo'q."
            )
            return

        lines = [
            "<b>🧠 Sening xotiralaring</b>",
            "",
        ]

        for memory in memories:
            lines.append(format_memory(memory))
            lines.append("")

        lines.append(
            "O'chirish: <code>/forget ID</code>"
        )

        await message.answer(
            "\n".join(lines),
            parse_mode="HTML",
        )

        return

    # --------------------------------------------------------
    # /memory search TEXT
    # --------------------------------------------------------

    parts = args.split(maxsplit=1)

    action = parts[0].lower()

    if action == "search":

        if len(parts) < 2 or not parts[1].strip():
            await message.answer(
                "🔎 Qidirish uchun matn yoz.\n\n"
                "Misol:\n"
                "<code>/memory search minecraft</code>"
            )
            return

        query = parts[1].strip()

        memories = await memory_manager.search_user_memories(
            user_telegram_id=message.from_user.id,
            search_text=query,
            limit=settings.max_memory_results,
        )

        if not memories:
            await message.answer(
                f"🔎 <b>{query}</b> bo'yicha xotira topilmadi.",
                parse_mode="HTML",
            )
            return

        lines = [
            f"🔎 <b>Qidiruv:</b> {query}",
            "",
        ]

        for memory in memories:
            lines.append(format_memory(memory))
            lines.append("")

        await message.answer(
            "\n".join(lines),
            parse_mode="HTML",
        )

        return

    # --------------------------------------------------------
    # /memory stats
    # --------------------------------------------------------

    if action == "stats":

        count = await memory_manager.user_memory_count(
            user_telegram_id=message.from_user.id
        )

        await message.answer(
            "📊 <b>Memory Statistics</b>\n\n"
            f"🧠 Saqlangan memory: <b>{count}</b>\n"
            f"📦 Maksimal ko'rsatiladigan: "
            f"<b>{settings.max_memory_results}</b>\n"
            f"💾 Memory system: "
            f"<b>{'ON' if settings.memory_enabled else 'OFF'}</b>",
            parse_mode="HTML",
        )

        return

    # --------------------------------------------------------
    # Noma'lum action
    # --------------------------------------------------------

    await message.answer(
        "❓ Noma'lum memory command.\n\n"
        "Foydalanish:\n"
        "<code>/memory</code>\n"
        "<code>/memory list</code>\n"
        "<code>/memory search matn</code>\n"
        "<code>/memory stats</code>",
        parse_mode="HTML",
    )

@router.message(Command("remind"))
async def command_remind(message: Message) -> None:
    if message.from_user is None:
        return

    text = get_command_args(message)

    if not text:
        await message.answer(
            "⏰ Reminder matnini yoz.\n\n"
            "Misol:\n"
            "<code>/remind 10 daqiqadan keyin menga yoz</code>\n\n"
            "Yoki:\n"
            "<code>/remind tomorrow 10:00 loyiha haqida eslat</code>",
            parse_mode="HTML",
        )
        return

    try:
        reminder = await create_reminder_from_text(
            owner_telegram_id=message.from_user.id,
            chat_id=message.chat.id,
            text=text,
        )

        await message.answer(
            "⏰ <b>Reminder yaratildi!</b>\n\n"
            f"📝 {reminder.text}\n"
            f"🕐 <code>{reminder.remind_at}</code>\n"
            f"🆔 <code>{reminder.id}</code>",
            parse_mode="HTML",
        )

    except ValueError as exc:
        await message.answer(
            f"❌ {str(exc)}"
        )

    except Exception:
        logger.exception(
            "Reminder yaratishda xato | user=%s",
            message.from_user.id,
        )

        await message.answer(
            "❌ Reminder yaratishda xatolik yuz berdi."
        )

@router.message(Command("reminders"))

        owner_telegram_id=message.from_user.id

        )

    lines = [
        "⏰ <b>Sening reminderlaring</b>",
        "",

    for reminder in reminders:

        lines.append(

            f"📝 {reminder.text}\n"
            f"🕐 <code>{reminder.remind_at}</code>"

    lines.append(
        "Bekor qilish:\n"
        "<code>/cancel_reminder ID</code>"

        "\n".join(lines),
