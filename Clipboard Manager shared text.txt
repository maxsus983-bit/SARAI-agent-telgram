# ============================================================
# HELPERS
# ============================================================


def get_args(message: Message) -> str:
    text = message.text or ""

    parts = text.split(maxsplit=1)

    if len(parts) < 2:
        return ""

    return parts[1].strip()


def get_args_list(message: Message) -> list[str]:
    args = get_args(message)

    if not args:
        return []

    return args.split()


async def is_group_admin(message: Message) -> bool:
    if message.chat.type not in {"group", "supergroup"}:
        return False

    if message.from_user is None:
        return False

    return await permission_service.can_manage_group_async(
        bot=message.bot,
        chat_id=message.chat.id,
        user_id=message.from_user.id,
    )


def memory_to_text(memory) -> str:
    return (
        f"🧠 <b>#{memory.id}</b>\n"
        f"📌 <code>{memory.memory_type}</code>\n"
        f"⭐ {memory.importance}/100\n"
        f"🎯 {memory.confidence:.2f}\n"
        f"📝 {memory.content}"
    )


def group_memory_to_text(memory) -> str:
    return (
        f"👥 <b>#{memory.id}</b>\n"
        f"📌 <code>{memory.memory_type}</code>\n"
        f"⭐ {memory.importance}/100\n"

@router.message(Command("memory"))
async def memory_command(message: Message) -> None:
    if message.from_user is None:
        return

    args = get_args(message)

    if not args or args.lower() == "list":
        memories = await memory_manager.get_user_memories(
            user_telegram_id=message.from_user.id,
            limit=settings.max_memory_results,
        )

        if not memories:
            await message.answer(
                "🧠 Hozircha saqlangan memory yo'q."
            )
            return

        lines = ["<b>🧠 Sening memorylaring</b>", ""]

        for memory in memories:
            lines.append(memory_to_text(memory))
            lines.append("")

        lines.append(
            "O'chirish: <code>/forget ID</code>"
        )

        await message.answer(
            "\n".join(lines),
            parse_mode="HTML",
        )

        return

    parts = args.split(maxsplit=1)

    action = parts[0].lower()

    if action == "search":
        if len(parts) < 2:
            await message.answer(
                "🔎 Misol:\n"
                "<code>/memory search Minecraft</code>",
                parse_mode="HTML",
            )
            return

        memories = await memory_manager.search_user_memories(
            user_telegram_id=message.from_user.id,
            search_text=parts[1],
            limit=settings.max_memory_results,
        )

        if not memories:
            await message.answer(
                "🔎 Memory topilmadi."
            )
            return

        lines = [
            f"🔎 <b>{parts[1]}</b>",
            "",
        ]

        for memory in memories:
            lines.append(memory_to_text(memory))
            lines.append("")

        await message.answer(
            "\n".join(lines),
            parse_mode="HTML",
        )

        return

    if action == "stats":
        count = await memory_manager.user_memory_count(
            user_telegram_id=message.from_user.id
        )

        await message.answer(
            "📊 <b>MEMORY STATS</b>\n\n"
            f"🧠 Memory: <b>{count}</b>\n"
            f"💾 System: "
            f"<b>{'ON' if settings.memory_enabled else 'OFF'}</b>",
            parse_mode="HTML",
        )

        return

    await message.answer(
        "❓ Noma'lum memory command.\n\n"
        "<code>/memory</code>\n"
        "<code>/memory search text</code>\n"
        "<code>/memory stats</code>",
        parse_mode="HTML",
    )

@router.message(Command("forget"))

            "Misol: <code>/forget 12</code>",

        )

    except ValueError:
        await message.answer(
            "❌ ID raqam bo'lishi kerak."
        )

        user_telegram_id=message.from_user.id,
        memory_id=memory_id,
    )

    if success:
        await message.answer(
            f"🗑 <b>#{memory_id}</b> o'chirildi.",
            parse_mode="HTML",
        )
    else:
        await message.answer(
            "❌ Memory topilmadi yoki sizga tegishli emas."
        )

@router.message(Command("group_memory"))
async def group_memory_command(message: Message) -> None:
    if message.chat.type not in {"group", "supergroup"}:
        await message.answer(
            "👥 Bu command faqat guruhda ishlaydi."
        )
        return

    args = get_args(message)

    if args.lower().startswith("search "):
        query = args[7:].strip()

        memories = await memory_manager.search_group_memories(
            group_telegram_id=message.chat.id,
            search_text=query,
            limit=settings.max_memory_results,
        )

        if not memories:
            await message.answer(
                "🔎 Guruh memorysidan hech narsa topilmadi."
            )
            return

        lines = [
            f"👥🔎 <b>{query}</b>",
            "",
        ]

        for memory in memories:
            lines.append(group_memory_to_text(memory))
            lines.append("")

        await message.answer(
            "\n".join(lines),
            parse_mode="HTML",
        )

        return

    memories = await memory_manager.get_group_memories(
        group_telegram_id=message.chat.id,
        limit=settings.max_memory_results,
    )

    if not memories:
        await message.answer(
            "👥 Hozircha guruh memorysi yo'q."
        )
        return

    lines = [
        "👥 <b>GROUP MEMORY</b>",
        "",
    ]

    for memory in memories:
        lines.append(group_memory_to_text(memory))
        lines.append("")

    await message.answer(
        "\n".join(lines),
        parse_mode="HTML",
    )

@router.message(Command("remind"))
async def remind_command(message: Message) -> None:
    if message.from_user is None:
        return

    text = get_args(message)

    if not text:
        await message.answer(
            "⏰ Misol:\n"
            "<code>/remind 10 daqiqadan keyin menga yoz</code>",
            parse_mode="HTML",
        )
        return

    try:
        reminder = await create_reminder_from_text(
            owner_telegram_id=message.from_user.id,
            chat_id=message.chat.id,
            text=text,
        )

        await message.answer(
            "⏰ <b>Reminder yaratildi!</b>\n\n"
            f"📝 {reminder.text}\n"
            f"🕐 <code>{reminder.remind_at}</code>\n"
            f"🆔 <code>{reminder.id}</code>",
            parse_mode="HTML",
        )

    except ValueError as exc:
        await message.answer(
            f"❌ {exc}"
        )

    except Exception:
        logger.exception("Reminder yaratishda xato.")

        await message.answer(
            "❌ Reminder yaratishda xatolik."
        )
